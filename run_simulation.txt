ubuntu terminal 1 - start gazebo: 
    gz sim gazebo_sim/mecanum_world.sdf

ubuntu terminal 2 - run bridge:  
    ros2 run ros_gz_bridge parameter_bridge \
    '/cmd_vel@geometry_msgs/msg/Twist]gz.msgs.Twist' \
    '/model/mecanum_robot/odometry@nav_msgs/msg/Odometry[gz.msgs.Odometry'

ubuntu terminal 3 - run the receiver:
    python3 gazebo_sim/gazebo_receiver.py

powershell terminal - run python ui:
    python python/controller_gui_gazebo.py
